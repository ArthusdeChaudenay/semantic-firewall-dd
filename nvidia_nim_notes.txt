from fastapi import BackgroundTasks, UploadFile, File
import requests

# (Assure-toi que certify_document_from_bytes est bien importé depuis pipeline.py)

def send_workflow_alert(filename: str, report: dict):
    """
    Fonction qui pousse l'alerte vers Teams, Slack ou Airtable.
    Elle est exécutée en tâche de fond.
    """
    # Si le document est parfait, on ne spamme pas les analystes
    if report.get("statut") == "CERTIFIÉ":
        print(f"✅ {filename} est clean. Aucune alerte envoyée.")
        return

    # Extraction des erreurs détectées par ton pare-feu
    anomalies = report.get("anomalies", ["Dérive sémantique ou erreur inconnue"])
    anomalies_texte = "\n- ".join(anomalies)

    # Exemple de payload pour Microsoft Teams (via Webhook entrant)
    webhook_url = "TON_LIEN_WEBHOOK_TEAMS_ICI" # À remplacer par le vrai lien
    
    message = {
        "title": f"🚨 Alerte Due Diligence Pare-feu : {filename}",
        "text": f"Le modèle a bloqué ce document.\n\n**Anomalies détectées :**\n- {anomalies_texte}"
    }
    
    # Envoi de la requête HTTP au workflow
    try:
        # requests.post(webhook_url, json=message) # Décommente pour la vraie prod
        print(f"--> [SIMULATION RÉSEAU] Alerte envoyée au workflow pour {filename}")
        print(message)
    except Exception as e:
        print(f"Erreur lors de l'envoi de l'alerte : {e}")

@app.post("/webhook/ingest", tags=["Connecteur Réseau"])
async def ingest_from_vdr(background_tasks: BackgroundTasks, file: UploadFile = File(...)):
    """
    Endpoint appelé automatiquement par la VDR ou SharePoint 
    dès qu'un nouveau fichier financier est déposé.
    """
    # 1. Lecture du fichier en mémoire
    content = await file.read()
    
    # 2. Passage dans le Pare-feu Sémantique
    report = certify_document_from_bytes(content, file.filename)
    
    # 3. Déclenchement de l'alerte de manière asynchrone
    background_tasks.add_task(send_workflow_alert, file.filename, report)
    
    return {
        "message": "Fichier intercepté avec succès. L'analyse est terminée et l'alerte a été routée.",
        "fichier": file.filename,
        "statut_parefeu": report.get("statut")
    }
